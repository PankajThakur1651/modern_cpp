# Register every executable in the project as a CTest smoke test.
function(add_smoke_tests dir)
    get_property(targets DIRECTORY ${dir} PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS targets)
        get_target_property(type ${target} TYPE)
        if(type STREQUAL "EXECUTABLE")
            add_test(NAME ${target}_smoke COMMAND ${target})
            set_tests_properties(${target}_smoke PROPERTIES TIMEOUT 20)
        endif()
    endforeach()
    get_property(subdirs DIRECTORY ${dir} PROPERTY SUBDIRECTORIES)
    foreach(subdir IN LISTS subdirs)
        add_smoke_tests(${subdir})
    endforeach()
endfunction()

if(BUILD_TESTING)
    add_smoke_tests(${CMAKE_SOURCE_DIR})
endif()
